# Medlange Trainer's dependency set, pinned exactly.
#
# WHAT THIS INSTALLATION IS
# --------------------------
# A vanilla-PyTorch framework for volumetric medical image segmentation,
# written from scratch. There is NO nnU-Net and NO MONAI anywhere in this
# tree: the framework carries its own 3D UNet, planner, masked-loss training
# loop and sliding-window inference under `medos_trainer/vanilla/`. Every pin
# below is a dependency of the trainer and of nothing else.
#
# WHY THIS FILE IS NOT `deploy/compose/requirements.txt`
# -------------------------------------------------------
# `deploy/compose/requirements.txt` serves the platform image; this file
# serves the trainer. The two sets must stay separate: the trainer is the
# only tree that imports torch, and the platform image must be installable
# and servable without it.
#
# TORCH CARRIES THE CUDA USERSPACE, AND THAT IS DELIBERATE
# ---------------------------------------------------------
# `torch==2.7.1+cu128` declares `nvidia-cuda-runtime-cu12`,
# `nvidia-cudnn-cu12`, `nvidia-cublas-cu12` and the rest as its own pinned
# wheels, so every CUDA library in this image is pinned by the same resolver
# that pinned torch, and `pip freeze` is the complete answer to "what
# touched the data". An `apt-get install cuda-*` on top of an `nvidia/cuda`
# base would put a second, apt-pinned copy of the same libraries in the
# image and leave the loader to choose -- a reproducibility hole in the one
# record a training run's provenance exists to make trustworthy. The driver
# is not in the image and must not be: it is injected by the container
# runtime from the host.
#
# THE PIN STAYS EXACT, for the reason it was exact before: a recorded
# version takes no compatibility range, and the environment declaration
# this image emits records exact versions.
--extra-index-url https://download.pytorch.org/whl/cu128

torch==2.7.1+cu128

# numpy is the array format the whole stack speaks: `.npz` cases, batches,
# plans and bundles all round-trip through it. Pinned exact, same reason.
numpy==1.26.4

# scipy backs the detection/overlap metric utilities
# (`medos_trainer/detection.py`, `medos_trainer/overlap.py`) -- connected
# components and distance transforms over label arrays.
scipy==1.14.1

# nibabel is the ONE optional dependency: `vanilla-import-nnunet` reads
# NIfTI with it, imported lazily inside the one function that needs it, so
# an environment without it pays nothing and every other subcommand keeps
# working. Pinned because the importer test round-trips real NIfTI bytes.
nibabel==5.2.1
